enum ThreatLevel {
    NONE, LOW, MEDIUM, HIGH, CRITICAL
}

class CorruptedLogException extends Exception {
    public CorruptedLogException(String message) {
        super(message);
    }
}

class MaliciousPayloadException extends Exception {
    private final String sourceIp;
    private final String payload;

    public MaliciousPayloadException(String message, String sourceIp, String payload) {
        super(message);
        this.sourceIp = sourceIp;
        this.payload = payload;
    }

    public String getSourceIp() { return sourceIp; }
    public String getPayload() { return payload; }
}

class NetworkPacket {
    private final String sourceIp;
    private final int destinationPort;
    private final String payload;

    public NetworkPacket(String sourceIp, int destinationPort, String payload) {
        this.sourceIp = sourceIp;
        this.destinationPort = destinationPort;
        this.payload = payload;
    }

    public String getSourceIp() { return sourceIp; }
    public int getDestinationPort() { return destinationPort; }
    public String getPayload() { return payload; }
}

class InspectionResult {
    private final boolean passed;
    private final ThreatLevel threatLevel;
    private final String reason;

    public InspectionResult(boolean passed, ThreatLevel threatLevel, String reason) {
        this.passed = passed;
        this.threatLevel = threatLevel;
        this.reason = reason;
    }

    public boolean isPassed() { return passed; }
    public ThreatLevel getThreatLevel() { return threatLevel; }
    public String getReason() { return reason; }
}

class EvaluatedRecord {
    NetworkPacket packet;
    String status;
    String severity;

    public EvaluatedRecord(NetworkPacket packet, String status, String severity) {
        this.packet = packet;
        this.status = status;
        this.severity = severity;
    }
}

interface SecurityRule {
    InspectionResult evaluate(NetworkPacket packet) throws MaliciousPayloadException;
}

class IpBlacklistRule implements SecurityRule {
    private final List<String> blacklistedIps;

    public IpBlacklistRule(List<String> blacklistedIps) {
        this.blacklistedIps = blacklistedIps;
    }

    @Override
    public InspectionResult evaluate(NetworkPacket packet) {
        if (blacklistedIps.contains(packet.getSourceIp())) {
            return new InspectionResult(false, ThreatLevel.MEDIUM, "Blacklisted IP address");
        }
        return new InspectionResult(true, ThreatLevel.NONE, "IP clean");
    }
}

class PortFilterRule implements SecurityRule {
    private final List<Integer> blockedPorts;

    public PortFilterRule(List<Integer> blockedPorts) {
        this.blockedPorts = blockedPorts;
    }

    @Override
    public InspectionResult evaluate(NetworkPacket packet) {
        if (blockedPorts.contains(packet.getDestinationPort())) {
            return new InspectionResult(false, ThreatLevel.LOW, "Blocked port " + packet.getDestinationPort());
        }
        return new InspectionResult(true, ThreatLevel.NONE, "Port clean");
    }
}

class PayloadSignatureRule implements SecurityRule {
    private final List<String> attackSignatures;

    public PayloadSignatureRule(List<String> attackSignatures) {
        this.attackSignatures = attackSignatures;
    }

    @Override
    public InspectionResult evaluate(NetworkPacket packet) throws MaliciousPayloadException {
        String content = packet.getPayload().toUpperCase();
        for (String sig : attackSignatures) {
            if (content.contains(sig.toUpperCase())) {
                throw new MaliciousPayloadException("Critical exploit detected: " + sig, packet.getSourceIp(), packet.getPayload());
            }
        }
        return new InspectionResult(true, ThreatLevel.NONE, "Payload clean");
    }
}

class FirewallEngine {
    private final List<SecurityRule> activeRules = new ArrayList<>();

    public void addRule(SecurityRule rule) {
        activeRules.add(rule);
    }

    public InspectionResult inspect(NetworkPacket packet) throws MaliciousPayloadException {
        for (SecurityRule rule : activeRules) {
            InspectionResult result = rule.evaluate(packet);
            if (!result.isPassed()) {
                return result;
            }
        }
        return new InspectionResult(true, ThreatLevel.NONE, "Passed all rules");
    }
}

        b.setMaximumSize(new Dimension(0, 0));
        b.setBorder(BorderFactory.createEmptyBorder());

        b.setFocusable(false);
        return b;
    }

    @Override
    public Dimension getPreferredSize(JComponent c) {
        return scrollbar.getOrientation() == Adjustable.VERTICAL
                ? new Dimension(THICKNESS, 0)
                : new Dimension(0, THICKNESS);
    }

    @Override
    protected void paintTrack(Graphics g, JComponent c, Rectangle r) {
        // Transparent track: the card/background shows through.
    }

    @Override
    protected void paintThumb(Graphics g, JComponent c, Rectangle r) {
        if (r.isEmpty() || !scrollbar.isEnabled()) return;
        Graphics2D g2 = (Graphics2D) g.create();
        g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);

        Color base = isDragging ? Theme.COLOR_PRIMARY
                : isThumbRollover() ? Theme.mix(Theme.EDGE, Theme.COLOR_PRIMARY, 0.55f)
                  : Theme.EDGE;

        int pad = 2;
        boolean vert = scrollbar.getOrientation() == Adjustable.VERTICAL;
        float x = r.x + (vert ? pad : 0), y = r.y + (vert ? 0 : pad);
        float w = r.width - (vert ? pad * 2 : 0), h = r.height - (vert ? 0 : pad * 2);
        float arc = Math.min(w, h);

        g2.setColor(base);
        g2.fill(new RoundRectangle2D.Float(x, y, w, h, arc, arc));
        g2.dispose();
    }
}

/**
 * FlowLayout that wraps components onto new rows and reports the correct
 * preferred height, so the container grows/shrinks as the window is resized.
 */
class WrapLayout extends FlowLayout {
    public WrapLayout(int align, int hgap, int vgap) {
        super(align, hgap, vgap);
    }

    @Override
    public Dimension preferredLayoutSize(Container target) {
        return layoutSize(target, true);
    }

    @Override
    public Dimension minimumLayoutSize(Container target) {
        Dimension minimum = layoutSize(target, false);
        minimum.width -= (getHgap() + 1);
        return minimum;
    }

    private Dimension layoutSize(Container target, boolean preferred) {
        synchronized (target.getTreeLock()) {
            Container container = target;
            while (container.getSize().width == 0 && container.getParent() != null) {
                container = container.getParent();
            }
            int targetWidth = container.getSize().width;
            if (targetWidth == 0) targetWidth = Integer.MAX_VALUE;

            int hgap = getHgap();
            int vgap = getVgap();
            Insets insets = target.getInsets();
            int horizontalInsetsAndGap = insets.left + insets.right + (hgap * 2);
            int maxWidth = targetWidth - horizontalInsetsAndGap;

            Dimension dim = new Dimension(0, 0);
            int rowWidth = 0;
            int rowHeight = 0;

            for (int i = 0; i < target.getComponentCount(); i++) {
                Component m = target.getComponent(i);
                if (!m.isVisible()) continue;
                Dimension d = preferred ? m.getPreferredSize() : m.getMinimumSize();
                if (rowWidth + d.width > maxWidth && rowWidth > 0) {
                    addRow(dim, rowWidth, rowHeight);
                    rowWidth = 0;
                    rowHeight = 0;
                }
                if (rowWidth != 0) rowWidth += hgap;
                rowWidth += d.width;
                rowHeight = Math.max(rowHeight, d.height);
            }
            addRow(dim, rowWidth, rowHeight);

            dim.width += horizontalInsetsAndGap;
            dim.height += insets.top + insets.bottom + vgap * 2;
            return dim;
        }
    }

    private void addRow(Dimension dim, int rowWidth, int rowHeight) {
        dim.width = Math.max(dim.width, rowWidth);
        if (dim.height > 0) dim.height += getVgap();
        dim.height += rowHeight;
    }
}

/** Panel that follows the viewport width, so it can live inside a scroll pane. */
class ScrollablePanel extends JPanel implements Scrollable {
    @Override public Dimension getPreferredScrollableViewportSize() { return getPreferredSize(); }
    @Override public int getScrollableUnitIncrement(Rectangle r, int o, int d) { return 16; }
    @Override public int getScrollableBlockIncrement(Rectangle r, int o, int d) { return 64; }
    @Override public boolean getScrollableTracksViewportWidth() { return true; }

    @Override
    public boolean getScrollableTracksViewportHeight() {
        // Stretch to fill a tall viewport; scroll when the viewport is shorter.
        return getParent() instanceof JViewport
                && getParent().getHeight() > getPreferredSize().height;
    }
}

/** Soft aubergine gradient window background with lavender and rose glows. */
class GradientPanel extends JPanel {
    public GradientPanel() {
        setOpaque(true);
    }

    @Override
    protected void paintComponent(Graphics g) {
        Graphics2D g2 = (Graphics2D) g.create();
        g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
        int w = getWidth(), h = getHeight();

        g2.setPaint(new GradientPaint(0, 0, new Color(20, 14, 26), w, h, new Color(40, 27, 49)));
        g2.fillRect(0, 0, w, h);

        float radius = Math.max(w, h) * 0.65f;
        g2.setPaint(new RadialGradientPaint(w * 0.25f, -30f, radius,
                new float[]{0f, 1f},
                new Color[]{new Color(198, 162, 228, 46), new Color(198, 162, 228, 0)}));
        g2.fillRect(0, 0, w, h);

        g2.setPaint(new RadialGradientPaint(w * 0.95f, h * 1.05f, radius * 0.8f,
                new float[]{0f, 1f},
                new Color[]{new Color(240, 118, 152, 34), new Color(240, 118, 152, 0)}));
        g2.fillRect(0, 0, w, h);

        g2.dispose();
    }
}

/** Rounded card with optional coloured accent strip. */
class RoundedPanel extends JPanel {
    private final Color fill;
    private final Color edge;
    private final int arc;
    private Color accent;

    public RoundedPanel(LayoutManager layout, Color fill, Color edge, int arc) {
        super(layout);
        this.fill = fill;
        this.edge = edge;
        this.arc = arc;
        setOpaque(false);
    }

    public void setAccent(Color accent) {

    }

    @Override

        int w = getWidth(), h = getHeight();

/**
 * Animated analytics card: block-rate donut, severity bars and top offenders.
 * The content is designed on a fixed "design height" and scaled down uniformly
 * when the panel is shorter, so nothing is ever clipped.
 */
class AnalyticsPanel extends RoundedPanel {
    private static final String[] NAMES = {"CLEAN", "LOW", "MEDIUM", "HIGH", "CRITICAL"};
    private static final String[] KEYS  = {"NONE", "LOW", "MEDIUM", "HIGH", "CRITICAL"};
    private static final Color[] COLORS = {
            Theme.COLOR_OK, Theme.COLOR_PRIMARY, Theme.COLOR_WARN,
            Theme.COLOR_HIGH, Theme.COLOR_DANGER
    };
    private static final float DESIGN_HEIGHT = 450f;

    private final float[] shown = new float[5];
    private final float[] target = new float[5];
    private final int[] counts = new int[5];
    private float rateShown = 0f, rateTarget = 0f;
    private int total = 0, blocked = 0;
    private List<Map.Entry<String, Integer>> topIps = new ArrayList<>();
    private final Timer anim;

    public AnalyticsPanel() {
        super(new BorderLayout(), Theme.BG_PANEL, Theme.EDGE, 30);
        setPreferredSize(new Dimension(310, 0));
        setMinimumSize(new Dimension(200, 0));
        anim = new Timer(16, e -> step());
    }

    /** Recalculate everything from the master record list. */
    public void update(List<EvaluatedRecord> records) {
        Map<String, Integer> ips = new HashMap<>();
        java.util.Arrays.fill(counts, 0);
        blocked = 0;
        total = records.size();

        for (EvaluatedRecord r : records) {
            for (int i = 0; i < KEYS.length; i++) {
                if (KEYS[i].equals(r.severity)) counts[i]++;
            }
            if (!"ALLOWED".equals(r.status)) {
                blocked++;
                ips.merge(r.packet.getSourceIp(), 1, Integer::sum);
            }
        }

        int max = 1;
        for (int c : counts) max = Math.max(max, c);
        for (int i = 0; i < 5; i++) target[i] = counts[i] / (float) max;
        rateTarget = total == 0 ? 0f : blocked / (float) total;

        topIps = new ArrayList<>(ips.entrySet());
        topIps.sort((a, b) -> b.getValue() - a.getValue());
        if (topIps.size() > 3) topIps = new ArrayList<>(topIps.subList(0, 3));

        anim.start();
    }

    private void step() {
        boolean settled = true;
        for (int i = 0; i < 5; i++) {
            float d = target[i] - shown[i];
            if (Math.abs(d) > 0.002f) { shown[i] += d * 0.18f; settled = false; }
            else shown[i] = target[i];
        }
        float dr = rateTarget - rateShown;
        if (Math.abs(dr) > 0.002f) { rateShown += dr * 0.18f; settled = false; }
        else rateShown = rateTarget;

        repaint();
        if (settled) anim.stop();
    }

    @Override
    protected void paintComponent(Graphics g) {
        super.paintComponent(g);
        Graphics2D g2 = (Graphics2D) g.create();
        g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
        g2.setRenderingHint(RenderingHints.KEY_TEXT_ANTIALIASING, RenderingHints.VALUE_TEXT_ANTIALIAS_ON);

        // Responsive scaling: shrink the whole drawing when the card is short.
        float scale = Math.max(0.55f, Math.min(1f, getHeight() / DESIGN_HEIGHT));
        g2.scale(scale, scale);

        int w = Math.round(getWidth() / scale), pad = 22;

        // Title
        g2.setFont(new Font("Segoe UI", Font.BOLD, 12));
        g2.setColor(Theme.COLOR_PRIMARY);
        g2.drawString("THREAT ANALYTICS", pad, 32);

        // Donut
        int d = 100, dx = pad, dy = 48;
        g2.setStroke(new BasicStroke(10f, BasicStroke.CAP_ROUND, BasicStroke.JOIN_ROUND));
        g2.setColor(Theme.BG_RAISED);
        g2.draw(new Ellipse2D.Float(dx + 7, dy + 7, d - 14, d - 14));
        if (rateShown > 0.001f) {
            g2.setColor(Theme.COLOR_DANGER);
            g2.draw(new Arc2D.Float(dx + 7, dy + 7, d - 14, d - 14, 90, -360f * rateShown, Arc2D.OPEN));
        }
        String pct = String.format("%.0f%%", rateShown * 100);
        g2.setFont(new Font("Consolas", Font.BOLD, 18));
        g2.setColor(Theme.TEXT);
        FontMetrics fm = g2.getFontMetrics();
        g2.drawString(pct, dx + (d - fm.stringWidth(pct)) / 2, dy + d / 2 + fm.getAscent() / 2 - 2);

        // Donut legend
        int lx = dx + d + 22;
        g2.setFont(new Font("Segoe UI", Font.BOLD, 10));
        g2.setColor(Theme.TEXT_MUTED);
        g2.drawString("BLOCKED", lx, dy + 26);
        g2.drawString("ALLOWED", lx, dy + 72);
        g2.setFont(new Font("Consolas", Font.BOLD, 22));
        g2.setColor(Theme.COLOR_DANGER);
        g2.drawString(String.valueOf(blocked), lx, dy + 48);
        g2.setColor(Theme.COLOR_OK);
        g2.drawString(String.valueOf(total - blocked), lx, dy + 94);

        // Severity bars
        int y0 = 176;
        g2.setFont(new Font("Segoe UI", Font.BOLD, 10));
        g2.setColor(Theme.TEXT_MUTED);
        g2.drawString("SEVERITY BREAKDOWN", pad, y0);

        int trackX = pad + 72;
        int trackW = Math.max(20, w - pad * 2 - 72 - 34);
        for (int i = 0; i < 5; i++) {
            int ry = y0 + 22 + i * 26;
            g2.setFont(new Font("Segoe UI", Font.BOLD, 10));
            g2.setColor(COLORS[i]);
            g2.drawString(NAMES[i], pad, ry + 8);

            g2.setColor(Theme.alpha(COLORS[i], 36));
            g2.fill(new RoundRectangle2D.Float(trackX, ry, trackW, 8, 8, 8));
            float fw = shown[i] * trackW;
            if (fw > 1f) {
                g2.setPaint(new GradientPaint(trackX, 0, Theme.alpha(COLORS[i], 130),
                        trackX + trackW, 0, COLORS[i]));
                g2.fill(new RoundRectangle2D.Float(trackX, ry, Math.max(fw, 8), 8, 8, 8));
            }

            g2.setFont(new Font("Consolas", Font.BOLD, 12));
            g2.setColor(Theme.TEXT);
            String cs = String.valueOf(counts[i]);
            g2.drawString(cs, w - pad - g2.getFontMetrics().stringWidth(cs), ry + 9);
        }

        // Top offenders
        int y1 = y0 + 22 + 5 * 26 + 16;
        g2.setFont(new Font("Segoe UI", Font.BOLD, 10));
        g2.setColor(Theme.TEXT_MUTED);
        g2.drawString("TOP OFFENDERS", pad, y1);

        if (topIps.isEmpty()) {
            g2.setFont(new Font("Segoe UI", Font.ITALIC, 12));
            g2.drawString("No blocked sources yet", pad, y1 + 24);
        } else {
            int best = topIps.get(0).getValue();
            for (int i = 0; i < topIps.size(); i++) {
                Map.Entry<String, Integer> en = topIps.get(i);
                int ry = y1 + 20 + i * 26;
                g2.setFont(new Font("Consolas", Font.PLAIN, 12));
                g2.setColor(Theme.TEXT);
                g2.drawString(en.getKey(), pad, ry + 9);

                int bx = pad + 112, bw = Math.max(20, w - pad * 2 - 112 - 28);
                g2.setColor(Theme.alpha(Theme.COLOR_DANGER, 36));
                g2.fill(new RoundRectangle2D.Float(bx, ry, bw, 6, 6, 6));
                g2.setColor(Theme.COLOR_DANGER);
                g2.fill(new RoundRectangle2D.Float(bx, ry, Math.max(6, bw * en.getValue() / (float) best), 6, 6, 6));

                g2.setFont(new Font("Consolas", Font.BOLD, 12));
                g2.setColor(Theme.TEXT);
                String cs = String.valueOf(en.getValue());
                g2.drawString(cs, w - pad - g2.getFontMetrics().stringWidth(cs), ry + 9);
            }
        }
        g2.dispose();
    }
}

/** Pill-shaped segmented selector (used for the speed and traffic filter). */
class SegmentedControl extends JComponent {
    private final String[] labels;
    private final Color accent;
    private final List<ActionListener> listeners = new ArrayList<>();
    private int selected = 0;
    private int hover = -1;

    public SegmentedControl(String[] labels, Color accent) {
        this.labels = labels;
        this.accent = accent;
        setCursor(new Cursor(Cursor.HAND_CURSOR));
        setPreferredSize(new Dimension(labels.length * 64, 34));
        setMinimumSize(new Dimension(labels.length * 40, 30));

        MouseAdapter ma = new MouseAdapter() {
            @Override
            public void mousePressed(MouseEvent e) {
                int i = indexAt(e.getX());
                if (i >= 0) setSelected(i);
            }

            @Override
            public void mouseMoved(MouseEvent e) {
                hover = indexAt(e.getX());
                repaint();
            }

            @Override
            public void mouseExited(MouseEvent e) {
                hover = -1;
                repaint();
            }
        };
        addMouseListener(ma);
        addMouseMotionListener(ma);
    }

    private int indexAt(int x) {
        if (getWidth() <= 0) return -1;
        return Math.max(0, Math.min(labels.length - 1, x * labels.length / getWidth()));
    }

    public int getSelected() {
        return selected;
    }

    public void setSelected(int index) {
        if (index == selected) return;
        selected = index;
        repaint();
        for (ActionListener l : listeners) {
            l.actionPerformed(new ActionEvent(this, ActionEvent.ACTION_PERFORMED, labels[index]));
        }
    }

    public void addActionListener(ActionListener l) {
        listeners.add(l);
    }

    @Override
    protected void paintComponent(Graphics g) {
        Graphics2D g2 = (Graphics2D) g.create();
        g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
        g2.setRenderingHint(RenderingHints.KEY_TEXT_ANTIALIASING, RenderingHints.VALUE_TEXT_ANTIALIAS_ON);
        int w = getWidth(), h = getHeight();

        g2.setColor(Theme.BG_CONSOLE);
        g2.fill(new RoundRectangle2D.Float(0, 0, w - 1, h - 1, h, h));
        g2.setColor(Theme.EDGE);
        g2.setStroke(new BasicStroke(1f));
        g2.draw(new RoundRectangle2D.Float(0, 0, w - 1, h - 1, h, h));

        float segW = (w - 6) / (float) labels.length;
        g2.setFont(new Font("Segoe UI", Font.BOLD, 11));
        FontMetrics fm = g2.getFontMetrics();

        for (int i = 0; i < labels.length; i++) {
            float x = 3 + i * segW;
            RoundRectangle2D.Float seg = new RoundRectangle2D.Float(x, 3, segW, h - 7, h, h);
            boolean sel = i == selected;

            if (sel) {
                g2.setPaint(new GradientPaint(0, 3, Theme.alpha(accent, 90), 0, h - 4, Theme.alpha(accent, 40)));
                g2.fill(seg);
                g2.setColor(Theme.alpha(accent, 170));
                g2.draw(seg);
            } else if (i == hover) {
                g2.setColor(Theme.alpha(accent, 24));
                g2.fill(seg);
            }

            g2.setColor(sel ? accent : (i == hover ? Theme.TEXT : Theme.TEXT_MUTED));
            String t = labels[i];
            g2.drawString(t, x + (segW - fm.stringWidth(t)) / 2f, (h + fm.getAscent() - fm.getDescent()) / 2f);
        }
        g2.dispose();
    }
}

    private final Timer timer;

    public ToastLayer() {
        setOpaque(false);
        timer = new Timer(30, e -> tick());
    }

    @Override
    public boolean contains(int x, int y) {
        return false; // never intercept the mouse
    }

    public void popup(String title, String message, Color color) {
        this.title = title;
        this.message = message;
        this.color = color;
        this.shownAt = System.currentTimeMillis();
        this.alpha = 0f;
        timer.restart();
    }

    private void tick() {
        long age = System.currentTimeMillis() - shownAt;
        if (age < 250) alpha = age / 250f;
        else if (age < 4250) alpha = 1f;
        else if (age < 4750) alpha = 1f - (age - 4250) / 500f;
        else { alpha = 0f; timer.stop(); }
        repaint();
    }

    @Override
    protected void paintComponent(Graphics g) {
        if (alpha <= 0.01f) return;
        Graphics2D g2 = (Graphics2D) g.create();
        g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
        g2.setRenderingHint(RenderingHints.KEY_TEXT_ANTIALIASING, RenderingHints.VALUE_TEXT_ANTIALIAS_ON);
        g2.setComposite(AlphaComposite.SrcOver.derive(Math.min(1f, alpha)));

        int pad = 16;
        int tw = Math.max(220, Math.min(430, getWidth() - 48));
        int th = 62;
        int x = getWidth() - tw - 24;
        int y = getHeight() - th - 24 + Math.round((1f - alpha) * 14);

        for (int i = 1; i <= 4; i++) {
            g2.setColor(new Color(0, 0, 0, 26 - i * 5));
            g2.fill(new RoundRectangle2D.Float(x - i, y - i + 3, tw + 2 * i, th + 2 * i, 22 + i * 2, 22 + i * 2));
        }

        RoundRectangle2D.Float card = new RoundRectangle2D.Float(x, y, tw, th, 22, 22);
        g2.setColor(Theme.BG_RAISED);
        g2.fill(card);

        Shape oldClip = g2.getClip();
        g2.setClip(card);
        g2.setColor(color);
        g2.fillRect(x, y, 5, th);
        g2.setClip(oldClip);

        g2.setColor(Theme.alpha(color, 140));
        g2.setStroke(new BasicStroke(1f));
        g2.draw(card);

        int tx = x + pad + 6;
        g2.setFont(new Font("Segoe UI", Font.BOLD, 13));
        g2.setColor(color);
        g2.drawString(title, tx, y + 25);

        g2.setFont(new Font("Consolas", Font.PLAIN, 12));
        g2.setColor(Theme.TEXT);
        FontMetrics fm = g2.getFontMetrics();
        int maxW = tw - pad * 2 - 6;
        String m = message;
        if (fm.stringWidth(m) > maxW) {
            while (m.length() > 4 && fm.stringWidth(m + "\u2026") > maxW) {
                m = m.substring(0, m.length() - 1);
            }
            m = m + "\u2026";
        }
        g2.drawString(m, tx, y + 47);
        g2.dispose();
    }
}

    /**
     * Breakpoints (by window width):
     *   tier 0 : >= 1280  full layout
     *   tier 1 : >= 1050  slightly narrower sidebar and analytics card
     *   tier 2 : >=  880  analytics card hidden, compact sidebar
     *   tier 3 : <   880  metric cards become a 2x2 grid, smallest fonts
     * Height below 720 switches to a compact "short window" mode.
     */
    private void applyResponsiveLayout() {
        int w = getWidth(), h = getHeight();
        if (w <= 0 || h <= 0 || sidebar == null) return;

        int tier = w >= 1280 ? 0 : (w >= 1050 ? 1 : (w >= 880 ? 2 : 3));
        boolean shortWindow = h < 720;

        // Cheap per-event updates (console height follows the window height)
        int consoleH = Math.max(70, Math.min(170, (int) (h * 0.16)));
        consoleScrollPane.setPreferredSize(new Dimension(10, consoleH));

        // Structural changes only when a breakpoint is crossed
        if (tier != lastTier || shortWindow != lastShortWindow) {
            lastTier = tier;
            lastShortWindow = shortWindow;

            int[] sidebarW = {262, 230, 205, 185};
            int[] analyticsW = {310, 270, 0, 0};
            int[] metricFont = {32, 28, 24, 22};
            int[] titleFont = {28, 26, 24, 22};
            int[] payloadCols = {24, 18, 14, 12};

            sidebar.setPreferredSize(new Dimension(sidebarW[tier], 0));
            sidebar.setBorder(BorderFactory.createEmptyBorder(
                    shortWindow ? 16 : 24, tier >= 2 ? 12 : 18, shortWindow ? 14 : 20, tier >= 2 ? 12 : 18));

            boolean showAnalytics = analyticsW[tier] > 0;
            analyticsPanel.setVisible(showAnalytics);
            if (showAnalytics) {
                analyticsPanel.setPreferredSize(new Dimension(analyticsW[tier], 0));
            }

            // Metric cards: 1 row of 4 on wide windows, 2x2 on narrow ones
            if (tier == 3) {
                metricsPanel.setLayout(new GridLayout(2, 2, 12, 12));
            } else {
                metricsPanel.setLayout(new GridLayout(1, 4, tier >= 2 ? 10 : 16, 12));
            }
            for (JLabel l : new JLabel[]{lblTotal, lblAllowed, lblBlocked, lblThreatRate}) {
                l.setFont(new Font("Consolas", Font.BOLD, shortWindow ? metricFont[tier] - 6 : metricFont[tier]));
            }

            hello.setFont(new Font("Segoe UI", Font.BOLD, titleFont[tier]));
            helloSub.setVisible(!shortWindow);

            txtPayload.setColumns(payloadCols[tier]);

            trafficTable.setRowHeight(shortWindow ? 26 : 34);
            tableCard.setPreferredSize(new Dimension(10, shortWindow ? 320 : 400));

            int segW = tier >= 2 ? 58 : 72;
            filterControl.setPreferredSize(new Dimension(segW * 3, 34));
            txtSearch.setColumns(tier == 0 ? 14 : (tier == 1 ? 11 : 8));

            // The Reason column only fits on wide windows
            boolean showReason = tier <= 1;
            boolean reasonShown = trafficTable.getColumnCount() == 6;
            if (showReason && !reasonShown) {
                trafficTable.getColumnModel().addColumn(reasonColumn);
            } else if (!showReason && reasonShown) {
                trafficTable.getColumnModel().removeColumn(reasonColumn);
            }
        }

        // Re-run layout; a second pass lets the wrapping injector row settle
        // after it learns its new width.
        getContentPane().revalidate();
        SwingUtilities.invokeLater(() -> {
            getContentPane().revalidate();
            getContentPane().repaint();
        });
    }

    }

    private void updateProgress() {
        int size = currentFileLines.size();
        if (size == 0) {
            progressStrip.setProgress(0f);
            tableTitle.setText("LIVE TRAFFIC");
            return;
        }
        progressStrip.setProgress(currentFileIndex / (float) size);
        tableTitle.setText("LIVE TRAFFIC   |   " + currentFileIndex + " / " + size);
    }

    private void applyTableFilter() {
        final int mode = filterControl.getSelected();
        final String query = txtSearch.getText().trim().toLowerCase();
        sorter.setRowFilter(new RowFilter<DefaultTableModel, Integer>() {
            @Override
            public boolean include(Entry<? extends DefaultTableModel, ? extends Integer> entry) {
                boolean allowed = "ALLOWED".equals(String.valueOf(entry.getValue(3)));
                if (mode == 1 && !allowed) return false;
                if (mode == 2 && allowed) return false;
                if (query.isEmpty()) return true;
                for (int i = 0; i < entry.getValueCount(); i++) {
                    if (String.valueOf(entry.getValue(i)).toLowerCase().contains(query)) return true;
                }
                return false;
            }
        });
    }

    private void initFileTimer() {
        fileInspectionTimer = new Timer(1000, e -> processNextFileLine());
    }

    private void startFileInspection() {
        String selectedFileName = (String) cbFileSelector.getSelectedItem();
        if (selectedFileName == null) return;

        String actualFileName = selectedFileName.split(" ")[0];
        currentFileLines = readLinesFromFile(actualFileName);
        currentFileIndex = 0;

        if (currentFileLines.isEmpty()) {
            JOptionPane.showMessageDialog(this, "The selected file is empty.", "File Error", JOptionPane.ERROR_MESSAGE);
            return;
        }

        isFileInspecting = true;
        isFilePaused = false;

        btnStartInspection.setEnabled(false);
        btnPause.setEnabled(true);
        btnResume.setEnabled(false);
        btnResumeFileStream.setVisible(false);

        lblStatusIndicator.setText("Checking " + actualFileName);
        lblStatusIndicator.setForeground(COLOR_OK);

        logConsole("Started checking " + actualFileName + " (" + currentFileLines.size() + " packets).");
        updateProgress();
        fileInspectionTimer.start();
    }

    private void pauseInspection() {
        if (!isFileInspecting || isFilePaused) return;

        isFilePaused = true;
        fileInspectionTimer.stop();

        btnPause.setEnabled(false);
        btnResume.setEnabled(true);

        lblStatusIndicator.setText("Paused");
        lblStatusIndicator.setForeground(COLOR_WARN);
        logConsole("Checking paused.");
    }

    private void resumeInspection() {
        if (!isFileInspecting || !isFilePaused) return;

        isFilePaused = false;
        fileInspectionTimer.start();

        btnPause.setEnabled(true);
        btnResume.setEnabled(false);
        btnResumeFileStream.setVisible(false);

        lblStatusIndicator.setText("Checking traffic");
        lblStatusIndicator.setForeground(COLOR_OK);
        logConsole("Checking resumed.");
    }

    private void processNextFileLine() {
        if (currentFileIndex < currentFileLines.size()) {
            String line = currentFileLines.get(currentFileIndex);
            currentFileIndex++;
            updateProgress();

            if (!line.trim().isEmpty() && !line.startsWith("#")) {
                try {
                    NetworkPacket packet = parseLogEntry(line);
                    evaluateAndDisplayPacket(packet);
                } catch (CorruptedLogException ex) {
                    logConsole("Skipped an invalid line: " + line);
                }
            }
        } else {
            fileInspectionTimer.stop();
            isFileInspecting = false;
            isFilePaused = false;

            btnStartInspection.setEnabled(true);
            btnPause.setEnabled(false);
            btnResume.setEnabled(false);

            lblStatusIndicator.setText("Finished");
            lblStatusIndicator.setForeground(COLOR_PRIMARY);

            logConsole("Finished checking the file (" + currentFileLines.size() + " records).");

            toastLayer.popup("Inspection finished",
                    currentFileLines.size() + " records checked. Pick another dataset.", COLOR_OK);
        }
    }

    private void injectManualPacketAndPauseFile() {
        String ip = txtIp.getText().trim();
        String portStr = txtPort.getText().trim();
        String payload = txtPayload.getText().trim();

        if (ip.isEmpty() || portStr.isEmpty() || payload.isEmpty()) {
            JOptionPane.showMessageDialog(this, "Please fill in all packet fields.", "Missing Information", JOptionPane.WARNING_MESSAGE);
            return;
        }

        try {
            int port = Integer.parseInt(portStr);

            if (isFileInspecting && !isFilePaused) {
                pauseInspection();
                btnResumeFileStream.setVisible(true);
                logConsole("File checking was paused so your packet could be tested.");
            }

            NetworkPacket packet = new NetworkPacket(ip, port, payload);
            evaluateAndDisplayPacket(packet);
            logConsole("Tested one packet from " + ip);

        } catch (NumberFormatException e) {
            logConsole("The port must be a number.");
            JOptionPane.showMessageDialog(this, "The port must be a number.", "Invalid Port", JOptionPane.ERROR_MESSAGE);
        }
    }

    private void resumeFileStream() {
        if (isFileInspecting && isFilePaused) {
            resumeInspection();
        }
    }

    private void evaluateAndDisplayPacket(NetworkPacket packet) {
        try {
            InspectionResult result = engine.inspect(packet);

            if (result.isPassed()) {
                masterRecordList.add(new EvaluatedRecord(packet, "ALLOWED", "NONE"));
                tableModel.insertRow(0, new Object[]{packet.getSourceIp(), packet.getDestinationPort(), packet.getPayload(), "ALLOWED", "NONE", "Passed all rules"});
                LogFileManager.logAllowed(packet);
                allowedCount++;
            } else {
                masterRecordList.add(new EvaluatedRecord(packet, "BLOCKED", result.getThreatLevel().toString()));
                tableModel.insertRow(0, new Object[]{packet.getSourceIp(), packet.getDestinationPort(), packet.getPayload(), "BLOCKED", result.getThreatLevel().toString(), result.getReason()});
                LogFileManager.logThreat(packet, result.getThreatLevel(), result.getReason());
                blockedCount++;
                logConsole("Blocked a packet from " + packet.getSourceIp() + " because of: " + result.getReason());
            }

        } catch (MaliciousPayloadException ex) {
            masterRecordList.add(new EvaluatedRecord(packet, "THREAT", "CRITICAL"));
            tableModel.insertRow(0, new Object[]{ex.getSourceIp(), packet.getDestinationPort(), ex.getPayload(), "THREAT", "CRITICAL", ex.getMessage()});
            LogFileManager.logThreat(packet, ThreatLevel.CRITICAL, ex.getMessage());
            blockedCount++;

            logConsole("Critical attack detected from " + ex.getSourceIp());

            toastLayer.popup("Critical attack blocked",
                    ex.getSourceIp() + "  -  " + ex.getMessage(), COLOR_DANGER);
        }

        totalCount++;
        updateMetrics();
    }

    private List<String> readLinesFromFile(String fileName) {
        List<String> lines = new ArrayList<>();
        File f = new File(fileName);
        if (!f.exists()) return lines;

        try (BufferedReader br = new BufferedReader(new FileReader(f))) {
            String l;
            while ((l = br.readLine()) != null) {
                lines.add(l);
            }
        } catch (IOException ignored) {}
        return lines;
    }

    private NetworkPacket parseLogEntry(String line) throws CorruptedLogException {
        String[] parts = line.split(",");
        if (parts.length < 3) {
            throw new CorruptedLogException("Invalid line format");
        }
        try {
            return new NetworkPacket(parts[0].trim(), Integer.parseInt(parts[1].trim()), parts[2].trim());
        } catch (NumberFormatException e) {
            throw new CorruptedLogException("Invalid port");
        }
    }

    private void updateMetrics() {
        lblTotal.setText(String.valueOf(totalCount));
        lblAllowed.setText(String.valueOf(allowedCount));
        lblBlocked.setText(String.valueOf(blockedCount));

        double rate = totalCount == 0 ? 0.0 : ((double) blockedCount / totalCount) * 100;
        lblThreatRate.setText(String.format("%.1f%%", rate));

        analyticsPanel.update(masterRecordList);
    }

    private void logConsole(String msg) {
        String time = LocalTime.now().format(DateTimeFormatter.ofPattern("HH:mm:ss"));
        consoleArea.append(time + "   " + msg + "\n");
        consoleArea.setCaretPosition(consoleArea.getDocument().getLength());
    }

    private void clearDashboard() {
        if (isFileInspecting) {
            fileInspectionTimer.stop();
            isFileInspecting = false;
            isFilePaused = false;
            btnStartInspection.setEnabled(true);
            btnPause.setEnabled(false);
            btnResume.setEnabled(false);
            btnResumeFileStream.setVisible(false);
        }
        masterRecordList.clear();
        tableModel.setRowCount(0);
        totalCount = 0;
        allowedCount = 0;
        blockedCount = 0;
        updateMetrics();
        consoleArea.setText("");
        lblStatusIndicator.setText("Idle");
        lblStatusIndicator.setForeground(Theme.TEXT_MUTED);
        currentFileIndex = 0;
        currentFileLines = new ArrayList<>();
        updateProgress();
        logConsole("Dashboard and counters were reset.");
    }
